3: // Deterministic guest clock used by the headless batch runner. Calls within a
4: // batch may advance the clock, but resetting the per-batch call counter must
5: // never make the next GetTickCount/timeGetTime call go backwards. A small
6: // reversal makes unsigned Win32 timer arithmetic look almost 25 days overdue.
7: function createBatchClock(tickMsPerBatch, tickCallStepMs = 1) {
8:   let batchStep = Math.max(0, Number(tickMsPerBatch) || 0);
9:   const callStep = Math.max(1, Number(tickCallStepMs) || 1);
10:   // `pausedMs` is guest time the harness has decided elapsed OUTSIDE the batch
11:   // schedule -- currently only `--dx-lock-pause-ms`, which charges a primary
12:   // surface Lock/Flip the display's frame interval so a run can be measured
13:   // with presentation back-pressure. It shifts the whole clock (base AND the
14:   // per-batch ceiling below) by the same amount, so it can never make the
15:   // clock go backwards and can never break the ceiling invariant. It is 0
16:   // unless a flag set it, and then it costs one add per call.
17:   const state = { batch: 0, callsInBatch: 0, lastTick: 0, pausedMs: 0 };
18:   let epochBatch = 0;
19:   let epochMs = 0;
20:
21:   const batchTicks = () => ((epochMs + (state.batch - epochBatch) * batchStep
22:     + state.pausedMs) | 0) & 0x7FFFFFFF;
23:
24:   const setTickMsPerBatch = value => {
25:     const next = Math.max(0, Number(value) || 0);
26:     // Rebase at the current batch before changing the slope. This preserves
27:     // the clock exactly at the transition instead of making a late switch
28:     // from (say) 200ms to 40ms jump backwards by batch * 160ms.
29:     epochMs = (Math.max(batchTicks(), state.lastTick) - state.pausedMs) | 0;
30:     epochBatch = state.batch;
31:     batchStep = next;
32:     return batchStep;
33:   };

35:   const getTicks = () => {
36:     const base = batchTicks();
37:     let candidate = ((base + (state.callsInBatch++ * callStep)) | 0) & 0x7FFFFFFF;
38:     // Calls may distinguish successive probes inside a batch, but they cannot
39:     // manufacture more time than the batch itself represents. Without this
40:     // ceiling, a tight PeekMessage/GetTickCount loop can keep a periodic timer
41:     // permanently due and never return to the application event queue.
